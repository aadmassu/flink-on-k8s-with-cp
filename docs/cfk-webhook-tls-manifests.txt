# CFK Operator Webhook TLS — cert-manager + PKCS12 Solution
# Validated on ROSA cluster (cfk-dev) — 2026-03-23
#
# Prerequisites:
#   - cert-manager installed
#   - Namespace: dah-kafka-dev (adjust as needed)
#
# How it works:
#   cert-manager creates a PKCS12 keystore (keystore.p12, truststore.p12)
#   A projected volume renames them to keystore.jks / truststore.jks
#   and merges in the jksPassword.txt file CFK expects.
#   The operator reads from /mnt/webhook-certs via directoryPathInContainer.

---
# 1. Self-signed Issuer
#    In production, replace with the KeyFactor ClusterIssuer.
apiVersion: cert-manager.io/v1
kind: Issuer
metadata:
  name: cfk-selfsigned-issuer
  namespace: dah-kafka-dev
spec:
  selfSigned: {}

---
# 2. Password secret for cert-manager's PKCS12 generation
#    In production, source from CyberArk via ExternalSecret.
apiVersion: v1
kind: Secret
metadata:
  name: cfk-jks-password
  namespace: dah-kafka-dev
stringData:
  password: changeit

---
# 3. Certificate CR — produces keystore.p12 and truststore.p12
#    cert-manager auto-renews 30 days before expiry.
apiVersion: cert-manager.io/v1
kind: Certificate
metadata:
  name: confluent-operator-serving-cert
  namespace: dah-kafka-dev
spec:
  secretName: confluent-operator-serving-cert
  issuerRef:
    name: cfk-selfsigned-issuer
    kind: Issuer
  dnsNames:
    - confluent-operator.dah-kafka-dev.svc
    - confluent-operator.dah-kafka-dev.svc.cluster.local
  duration: 8760h      # 1 year
  renewBefore: 720h    # 30 days
  keystores:
    pkcs12:
      create: true
      passwordSecretRef:
        name: cfk-jks-password
        key: password

---
# 4. Password file in CFK's expected format
#    CFK requires content: jksPassword=<value>
#    In production, source from CyberArk via ExternalSecret.
#    If stored as raw password in CyberArk, use ExternalSecret
#    templating to prepend "jksPassword=".
apiVersion: v1
kind: Secret
metadata:
  name: cfk-jks-password-file
  namespace: dah-kafka-dev
stringData:
  jksPassword.txt: "jksPassword=changeit"

---
# 5. Helm values for the CFK operator (relevant sections only)
#

#    └── jksPassword.txt   <- "jksPassword=changeit"
