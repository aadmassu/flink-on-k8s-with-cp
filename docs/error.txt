  Upload:
  $path = 'C:\path\to\uat.zip'
  $name = [System.IO.Path]::GetFileName($path)
  $b64  = [Convert]::ToBase64String([IO.File]::ReadAllBytes($path))
  Invoke-RestMethod -Uri 'https://text.psyncopate.io/api/files/upload' `
    -Method Post -ContentType 'application/json' `
    -Body (@{ name = $name; dataB64 = $b64 } | ConvertTo-Json)

  Download:
  $name = 'uat.zip'
  Invoke-WebRequest -Uri ("https://text.psyncopate.io/api/files/download?name=" + [uri]::EscapeDataString($name)) `
    -OutFile (".\" + $name)

  List:
  (Invoke-RestMethod -Uri 'https://text.psyncopate.io/api/files').files |
    Select-Object name, size, lastModified | Format-Table

  Delete:
  Invoke-RestMethod -Uri ("https://text.psyncopate.io/api/files?name=" + [uri]::EscapeDataString('uat.zip')) -Method Delete
